import { factories } from '@strapi/strapi';

export default factories.createCoreController(
  'api::order.order',
  ({ strapi }: { strapi: any }) => ({
    async find(ctx: any) {
      if (!ctx.state.user) {
        return ctx.unauthorized('You must be logged in to view orders.');
      }

      ctx.query = {
        ...ctx.query,
        filters: {
          ...ctx.query.filters,
          users_permissions_user: ctx.state.user.id,
        },
      };

      return super.find(ctx);
    },

    async findOne(ctx: any) {
      if (!ctx.state.user) {
        return ctx.unauthorized('You must be logged in to view this order.');
      }

      const order: any = await strapi.entityService.findOne('api::order.order', ctx.params.id, {
        populate: ['users_permissions_user'],
      });

      if (!order || order.users_permissions_user?.id !== ctx.state.user.id) {
        return ctx.notFound('Order not found.');
      }

      return super.findOne(ctx);
    },

    async create(ctx: any) {
      if (!ctx.state.user) {
        return ctx.unauthorized('You must be logged in to place an order.');
      }

      ctx.request.body.data = {
        ...ctx.request.body.data,
        users_permissions_user: ctx.state.user.id,
      };

      return super.create(ctx);
    },
  })
);
